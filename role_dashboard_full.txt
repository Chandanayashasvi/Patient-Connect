# PATIENT MANAGEMENT
# ============================================================

def patient_management(role):

    while True:

        print()
        print("========================================")
        print("        PATIENT MANAGEMENT")
        print("========================================")
        print("1. View Patients")
        print("2. Search Patient")
        print("3. View Clinics")
        print("4. View Providers")

        if role == "ADMIN":
            print("5. Create Patient")
            print("6. Back")
        else:
            print("5. Back")

        print("========================================")

        choice = input("Select option: ").strip()

        if choice == "1":
            patientconnect_cli.view_patients()

        elif choice == "2":
            patientconnect_cli.search_patient()

        elif choice == "3":
            patientconnect_cli.view_clinics()

        elif choice == "4":
            patientconnect_cli.view_providers()

        elif choice == "5" and role == "ADMIN":
            patientconnect_cli.create_patient()

        elif choice == "6" and role == "ADMIN":
            break

        elif choice == "5" and role != "ADMIN":
            break

        else:
            print("âŒ Invalid option.")

# ============================================================
# MAIN
# ============================================================

def main():

    user = login()

    if user is None:
        return

    role = str(user.get("role", "")).upper()

    print()
    print("========================================")
    print("       PATIENT CONNECT DASHBOARD")
    print("========================================")
    print("User :", user.get("username", "Unknown"))
    print("Role :", role)
    print("========================================")

    if role == "ADMIN":
        admin_dashboard(user)

    elif role == "DOCTOR":
        doctor_dashboard(user)

    elif role == "STAFF":
        staff_dashboard(user)

    else:
        print("âŒ Unknown role:", role)


# ============================================================
# START PROGRAM
# ============================================================

if __name__ == "__main__":
    main()
